Quick start

curl -fsSL https://forbear.dev/quick-start.sh | sh
irm https://www.forbear.dev/quick-start.ps1 | iex

The quick start asks for a language, Jai or C. Jai is in private beta, so if you don't have it, pick C. For C, you need MSVC on Windows, or gcc or clang on Linux and macOS.

forbear runs on Windows, Linux (Wayland) and macOS.

At a glance

Counter :: () {
    // State is stored in a 16MB static buffer with no heap allocations.
    counter := state(s64);

    if on_key_down(.PLUS) {
        // Since this is immediate mode, there's no need for callbacks/event handlers,
        // we can simply check for the key being down through an if-statement.
        counter.* += 1;
    } else if on_key_down(.MINUS) {
        counter.* -= 1;
    }

    element(Style.{
        // styling familiar to HTML
        background    = color(rgb(44, 49, 56)),
        color         = rgb(230, 234, 239),
        border_width  = all(1),
        border_color  = rgb(66, 73, 82),
        border_radius = 6,
        padding       = in_line(block(8), 16),
        margin        = block(8),
    }, #code { // The children of an element are defined through passing
        // around Code by the `#code` directive.
        node := get_node();
        if is_mouse_inside() {
            node.style.background = color(rgb(56, 62, 71));
            if is_mouse_held(.LEFT) || is_mouse_held(.RIGHT) {
                node.style.background = color(rgb(34, 38, 44));
            }
        }

        text("Click me");

        if on_click(.LEFT) {
            counter.* += 1;
        } else if on_click(.RIGHT) {
            counter.* -= 1;
        }
    });

    text(Style.{font_size = 32}, sprint("Count: %", counter.*));
} @component
void Counter(void) {
    // State is stored in a 16MB static buffer with no heap allocations.
    fb_s64 *counter = FB_STATE(fb_s64);

    if (fb_on_key_down(FB_KEY_PLUS, false)) {
        // Since this is immediate mode, there's no need for callbacks/event handlers,
        // we can simply check for the key being down through an if-statement.
        *counter += 1;
    } else if (fb_on_key_down(FB_KEY_MINUS, false)) {
        *counter -= 1;
    }

    FB_ELEMENT(
        // styling familiar to HTML
        .background    = fb_color(fb_rgb(44, 49, 56)),
        .color         = fb_rgb(230, 234, 239),
        .border_width  = fb_all_sides(1),
        .border_color  = fb_rgb(66, 73, 82),
        .border_radius = 6,
        .padding       = fb_with_in_line_sides(fb_block_sides(8), 16),
        .margin        = fb_block_sides(8),
    ) { // The children of an element go inside the block that follows FB_ELEMENT.
        fb_Node *node = fb_get_node(-1);
        if (fb_is_mouse_inside()) {
            node->style.background = fb_color(fb_rgb(56, 62, 71));
            if (fb_is_mouse_held(FB_MOUSE_BUTTON_LEFT) || fb_is_mouse_held(FB_MOUSE_BUTTON_RIGHT)) {
                node->style.background = fb_color(fb_rgb(34, 38, 44));
            }
        }

        FB_TEXT("Click me");

        if (fb_on_click(FB_MOUSE_BUTTON_LEFT, true, false)) {
            *counter += 1;
        } else if (fb_on_click(FB_MOUSE_BUTTON_RIGHT, true, false)) {
            *counter -= 1;
        }
    }

    char count_text[64];
    snprintf(count_text, sizeof(count_text), "Count: %lld", (long long)*counter);
    FB_TEXT(count_text, .font_size = 32);
}
// Call it as `FB_SCOPE Counter();` to give it its own state.
